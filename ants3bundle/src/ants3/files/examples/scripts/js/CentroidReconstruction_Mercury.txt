
// ------------- Configuration -------------
// Select the data directory
var dir = config.getKeyValue("PhotonSim.Run.OutputDirectory") // "/home/username/.config/ants3/TmpOutput"
var NumberOfEventMarkers = 100
// -----------------------------------------

var sensXYZ = sens.getSensorPositions()
if (sensXYZ.length == 0) core.abort("There are no optical sensors in th current config!")
response.newResponseModel(sensXYZ)
    
var sig = core.loadNumericArray(dir + "/SensorSignals.txt")
var numEv = sig.length
if (numEv == 0) core.abort("Failed to load sesnor signal data!")
core.print("Loaded", numEv, "events")

var truePos = core.loadNumericArray(dir + "/PhotonBombs.txt")
if (truePos.length != numEv) core.abort("Failed to load photon bomb positions!")

mercury.newReconstructor("COG", 15)
mercury.configure_COG(0,0,0)
mercury.setCutoffRadius(15)

core.print("Staring reconstruction");
var start = core.getTimeMarkMilliseconds()
mercury.reconstructEvents(sig)
var stop = core.getTimeMarkMilliseconds()
core.print("Elapsed time:", stop-start, "ms")

var recPosE = mercury.getRecXYZE()
core.print("Rec array size:", recPosE.length)

// Show event position markers
geowin.clearMarkers()
var arRec  = []
var arTrue = []
for (var iEv = 0; iEv < NumberOfEventMarkers; iEv++)
{
    arTrue.push(truePos[iEv])
    
    // Energy = 0 --> reconstruction failed
    if (recPosE[iEv][3] > 0) arRec.push(recPosE[iEv])
}
geowin.addMarkers(arTrue, 4, 7, 2)
geowin.addMarkers(arRec,  2, 2, 2)
geowin.showWindow()
