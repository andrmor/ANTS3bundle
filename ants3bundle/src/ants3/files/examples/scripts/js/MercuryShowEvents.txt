var dir = "/home/andr/WORK/ReconModule/Cases/Clinical/Out"
var showEvents = 10

var recPosE = mercury.getRecXYZE()

var truePos = core.loadNumericArray(dir + "/PhotonBombs.txt")
var numEv = recPosE.length
if (truePos.length != numEv) core.abort("Mismtach in reconstructed and reue position data!")

geowin.clearMarkers()
var arRec  = []
var arTrue = []
for (var iEv = 0; iEv < numEv && iEv < showEvents; iEv++)

    arTrue.push(truePos[iEv])
}
